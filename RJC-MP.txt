local function create(className, properties)
    local object = Instance.new(className)

    for property, value in pairs(properties or {}) do
        pcall(function()
            object[property] = value
        end)
    end

    return object
end

local function corner(parent, radius)
    return create("UICorner", {
        CornerRadius = UDim.new(0, radius or 7),
        Parent = parent,
    })
end

local function outline(parent, color, thickness)
    return create("UIStroke", {
        Color = color or COLORS.Border,
        Thickness = thickness or 1,
        Transparency = 0.18,
        Parent = parent,
    })
end

local function shorten(text, length)
    text = tostring(text or "")

    if #text <= length then
        return text
    end

    return text:sub(1, length - 3) .. "..."
end

local function formatNumber(number)
    number = tonumber(number)

    if not number then
        return "?"
    end

    local result = tostring(math.floor(number))

    while true do
        local replacement, count =
            result:gsub(
                "^(-?%d+)(%d%d%d)",
                "%1,%2"
            )

        result = replacement

        if count == 0 then
            break
        end
    end

    return result
end

local function copy(text)
    local methods = {
        function()
            if setclipboard then
                setclipboard(text)
                return true
            end
        end,

        function()
            if toclipboard then
                toclipboard(text)
                return true
            end
        end,

        function()
            if set_clipboard then
                set_clipboard(text)
                return true
            end
        end,
    }

    for _, method in ipairs(methods) do
        local ok, result = pcall(method)

        if ok and result then
            return true
        end
    end

    return false
end

local function getEnumItem(enumObject, candidates)
    if not enumObject then
        return nil
    end

    for _, name in ipairs(candidates) do
        local ok, value = pcall(function()
            return enumObject[name]
        end)

        if ok and value then
            return value
        end
    end

    return nil
end

local function assetType(name)
    return getEnumItem(
        Enum.AvatarAssetType,
        {name}
    )
end

local function bundleType(name)
    return getEnumItem(
        Enum.BundleType,
        {name}
    )
end

local function categoryFilter(name)
    return getEnumItem(
        Enum.CatalogCategoryFilter,
        {name}
    )
end

local function sortType(...)
    return getEnumItem(
        Enum.CatalogSortType,
        {
            ...
        }
    )
end

local function assets(...)
    local result = {}

    local names = {
        ...
    }

    for _, name in ipairs(names) do
        local value = assetType(name)

        if value then
            table.insert(result, value)
        end
    end

    return result
end

local function bundles(...)

    local names = {
        ...
    }

    for _, name in ipairs(names) do
        local value = bundleType(name)

        if value then
            table.insert(result, value)
        end
    end

    return result
end

local function closeDropdown()
    if activeDropdown then
        pcall(function()
            activeDropdown:Destroy()
        end)

        activeDropdown = nil
    end
end

-- DROPDOWN BUILDER
------------------------------------------------------------

local function makeDropdown(
    title,
    options,
    getIndex,
    setIndex
)

    local holder = create(
        "Frame",
        {
            BackgroundTransparency = 1,

            Size =
                UDim2.new(
                    1,
                    -2,
                    0,
                    43
                ),

            Parent =
                SidebarScroll,
        }
    )

    create(
        "TextLabel",
        {
            BackgroundTransparency = 1,

            Position =
                UDim2.fromOffset(
                    1,
                    0
                ),

            Size =
                UDim2.new(
                    1,
                    -2,
                    0,
                    12
                ),

            Font =
                Enum.Font.GothamBold,

            Text =
                title:upper(),

            TextColor3 =
                COLORS.SubText,

            TextSize = 7,

            TextXAlignment =
                Enum.TextXAlignment.Left,

            Parent =
                holder,
        }
    )

    local button = create(
        "TextButton",
        {
            BackgroundColor3 =
                COLORS.Panel2,

            BorderSizePixel = 0,

            Position =
                UDim2.fromOffset(
                    0,
                    14
                ),

            Size =
                UDim2.new(
                    1,
                    0,
                    0,
                    28
                ),

            Font =
                Enum.Font.Gotham,

            Text = "",

            TextColor3 =
                COLORS.Text,

            TextSize = 8,

            TextXAlignment =
                Enum.TextXAlignment.Left,

            AutoButtonColor = false,

            Parent =
                holder,
        }
    )

    corner(button, 6)
    outline(button)

    create(
        "TextLabel",
        {
            BackgroundTransparency = 1,

            Position =
                UDim2.new(
                    1,
                    -20,
                    0,
                    0
                ),

            Size =
                UDim2.fromOffset(
                    18,
                    28
                ),

            Font =
                Enum.Font.GothamBold,

            Text =
                "v",

            TextColor3 =
                COLORS.SubText,

            TextSize = 9,

            Parent =
                button,
        }
    )

    local function refresh()

        local index =
            getIndex()

        local option =
            options[index]

        if option then

            button.Text =
                "  "
                .. tostring(
                    option.name
                    or option[1]
                    or ""
                )
        end
    end

    button.Activated:Connect(function()

        closeDropdown()

        local menuHeight =
            math.min(
                220,
                8 + (#options * 27)
            )

        local menu = create(
            "ScrollingFrame",
            {
                BackgroundColor3 =
                    COLORS.Panel2,

                BorderSizePixel = 0,

                Size =
                    UDim2.fromOffset(
                        math.max(
                            120,
                            button.AbsoluteSize.X
                        ),
                        menuHeight
                    ),

                CanvasSize =
                    UDim2.new(
                        0,
                        0,
                        0,
                        0
                    ),

                AutomaticCanvasSize =
                    Enum.AutomaticSize.Y,

                ScrollBarThickness = 2,

                ScrollBarImageColor3 =
                    COLORS.Accent,

                ZIndex = 1000,

                Parent =
                    DropdownLayer,
            }
        )

        corner(menu, 7)
        outline(menu)

        activeDropdown = menu

        create(
            "UIPadding",
            {
                PaddingTop =
                    UDim.new(
                        0,
                        4
                    ),

                PaddingBottom =
                    UDim.new(
                        0,
                        4
                    ),

                Parent = menu,
            }
        )

        create(
            "UIListLayout",
            {
                Padding =
                    UDim.new(
                        0,
                        1
                    ),

                Parent = menu,
            }
        )

        for index, option in ipairs(options) do

            local optionButton =
                create(
                    "TextButton",
                    {
                        BackgroundColor3 =
                            index
                                == getIndex()
                            and COLORS.AccentDark
                            or COLORS.Panel2,

                        BackgroundTransparency =
                            index
                                == getIndex()
                            and 0
                            or 1,

                        BorderSizePixel = 0,

                        Size =
                            UDim2.new(
                                1,
                                -6,
                                0,
                                24
                            ),

                        Font =
                            Enum.Font.Gotham,

                        Text =
                            "  "
                            .. tostring(
                                option.name
                                or option[1]
                                or ""
                            ),

                        TextColor3 =
                            COLORS.Text,

                        TextSize = 8,

                        TextXAlignment =
                            Enum.TextXAlignment.Left,

                        AutoButtonColor = false,

                        ZIndex = 1001,

                        Parent = menu,
                    }
                )

            corner(optionButton, 5)

            optionButton.Activated:Connect(function()

                setIndex(
                    index,
                    option
                )

                refresh()

                closeDropdown()
            end)

            optionButton.MouseEnter:Connect(function()

                if index ~= getIndex() then

                    optionButton.BackgroundColor3 =
                        COLORS.Hover

                    optionButton.BackgroundTransparency = 0
                end
            end)

            optionButton.MouseLeave:Connect(function()

                if index ~= getIndex() then
                    optionButton.BackgroundTransparency = 1
                end
            end)
        end

        task.defer(function()

            if not menu.Parent then
                return
            end

            local cam =
                workspace.CurrentCamera

            local screenSize =
                cam
                and cam.ViewportSize
                or Vector2.new(
                    800,
                    600
                )

            local x =
                button.AbsolutePosition.X

            local y =
                button.AbsolutePosition.Y
                + button.AbsoluteSize.Y
                + 2

            local menuWidth =
                menu.AbsoluteSize.X

            local menuHeightActual =
                menu.AbsoluteSize.Y

            x =
                math.clamp(
                    x,
                    3,
                    math.max(
                        3,
                        screenSize.X
                        - menuWidth
                        - 3
                    )
                )

            y =
                math.clamp(
                    y,
                    3,
                    math.max(
                        3,
                        screenSize.Y
                        - menuHeightActual
                        - 3
                    )
                )

            menu.Position =
                UDim2.fromOffset(
                    x,
                    y
                )
        end)
    end)

    refresh()

    return holder
end

local ThumbnailQueue = {}
local ThumbnailWorkerRunning = false

local function thumbnailContentId(item)

    local id =
        tonumber(
            item.Id
            or item.id
        )

    if not id then
        return nil
    end

    local itemType =
        tostring(
            item.ItemType
            or item.itemType
            or ""
        ):lower()

    local isBundle =
        itemType == "bundle"
        or itemType == "2"

    if isBundle then

        return
            "rbxthumb://type=BundleThumbnail&id="
            .. tostring(id)
            .. "&w=420&h=420"
    end

    return
        "rbxthumb://type=Asset&id="
        .. tostring(id)
        .. "&w=420&h=420"
end

local function loadThumbnail(image, item)

    if not image
        or not image.Parent then
        return
    end

    local contentId =
        thumbnailContentId(
            item
        )

    if not contentId then

        image.Image = ""

        return
    end

    image.Image =
        contentId

    task.spawn(function()

        for attempt = 1,
            CONFIG.ThumbnailRetries do

            if not image.Parent then
                return
            end

            local success =
                pcall(function()

                    ContentProvider:PreloadAsync(
                        {
                            image
                        }
                    )
                end)

            if success then
                return
            end

            task.wait(
                CONFIG.ThumbnailRetryDelay
            )

            if image.Parent then
                image.Image =
                    contentId
            end
        end
    end)
end

local function startThumbnailWorker()

    if ThumbnailWorkerRunning then
        return
    end

    ThumbnailWorkerRunning = true

    task.spawn(function()

        while #ThumbnailQueue > 0 do

            local request =
                table.remove(
                    ThumbnailQueue,
                    1
                )

            if request
                and request.image
                and request.image.Parent then

                loadThumbnail(
                    request.image,
                    request.item
                )
            end

            task.wait(
                CONFIG.ThumbnailQueueDelay
            )
        end

        ThumbnailWorkerRunning = false
    end)
end

local function queueThumbnail(
    image,
    item
)

    table.insert(
        ThumbnailQueue,
        {
            image = image,
            item = item,
        }
    )

    startThumbnailWorker()
end

local function getPriceText(item)

    local priceStatus =
        tostring(
            item.PriceStatus
            or item.priceStatus
            or ""
        )

    if priceStatus == "Free" then
        return "FREE"
    end

    if priceStatus == "OffSale" then
        return "OFFSALE"
    end

    local lowestPrice =
        tonumber(
            item.LowestPrice
            or item.lowestPrice
        )

    if lowestPrice
        and lowestPrice > 0 then

        return
            "from "
            .. formatNumber(
                lowestPrice
            )
            .. " R$"
    end

    local price =
        tonumber(
            item.Price
            or item.price
        )

    if price then

        if price == 0 then
            return "FREE"
        end

        return
            formatNumber(
                price
            )
            .. " R$"
    end

    return "PRICE N/A"
end

-- BUILD SEARCH PARAMETERS
------------------------------------------------------------

local function buildSearchParameters()

    local params =
        CatalogSearchParams.new()

    --------------------------------------------------------
    -- Keyword
    --------------------------------------------------------

    params.SearchKeyword =
        tostring(
            SearchBox.Text
            or ""
        )

    --------------------------------------------------------
    -- Limit
    --------------------------------------------------------

    params.Limit =
        CONFIG.ResultsPerPage

    --------------------------------------------------------
    -- No offsale spam
    --------------------------------------------------------

    params.IncludeOffSale =
        false

    --------------------------------------------------------
    -- Category
    --------------------------------------------------------

    local category =
        CATEGORIES[
            state.categoryIndex
        ]

    if category then

        local filter =
            categoryFilter(
                category.category
            )

        if filter then
            params.CategoryFilter =
                filter
        end
    end

    --------------------------------------------------------
    -- Subcategory
    --------------------------------------------------------

    if category then

        local subcategory =
            category.subcategories[
                state.subcategoryIndex
            ]

        if subcategory then

            local assetTypes =
                subcategory.assetTypes

            local bundleTypes =
                subcategory.bundleTypes

            if assetTypes
                and #assetTypes > 0 then

                params.AssetTypes =
                    assetTypes
            end

            if bundleTypes
                and #bundleTypes > 0 then

                params.BundleTypes =
                    bundleTypes
            end
        end
    end

    --------------------------------------------------------
    -- Price
    --------------------------------------------------------

    local price =
        PRICES[
            state.priceIndex
        ]

    if price then

        if price.value == "custom" then

            local custom =
                tonumber(
                    CustomPrice.Text
                )

            if custom then

                params.MaxPrice =
                    math.max(
                        0,
                        math.floor(
                            custom
                        )
                    )
            end

        elseif price.value
            ~= nil then

            params.MaxPrice =
                price.value
        end
    end

    --------------------------------------------------------
    -- Sort
    --------------------------------------------------------

    local sort =
        SORTS[
            state.sortIndex
        ]

    if sort
        and sort.enum then

        params.SortType =
            sort.enum
    end

    --------------------------------------------------------
    -- Sales aggregation
    --------------------------------------------------------

    local currentSort =
        sort
        and sort.name
        or ""

    if currentSort ==
            "Most Favorited"

        or currentSort ==
            "Best Selling"
    then

        local allTime =
            getEnumItem(
                Enum.CatalogSortAggregation,
                {
                    "AllTime"
                }
            )

        if allTime then
            params.SortAggregation =
                allTime
        end
    end

    return params
end

local function searchMarketplace()

    if state.loading then
        return
    end

    state.loading = true

    closeDropdown()

    hideDetails()

    clearGrid()

    ResultStatus.Text =
        "Searching..."

    EmptyLabel.Visible =
        true

    EmptyLabel.Text =
        "Searching Marketplace..."

    state.pagesObject =
        nil

    state.pageCache =
        {}

    state.page =
        1

    local params

    local parameterOk,
        parameterResult =
        pcall(function()

            return buildSearchParameters()

        end)

    if not parameterOk then

        state.loading = false

        ResultStatus.Text =
            "Search setup failed"

        EmptyLabel.Text =
            tostring(
                parameterResult
            )

        return
    end

    params =
        parameterResult

    --------------------------------------------------------
    -- Native Roblox catalog search
    --------------------------------------------------------

    local searchOk,
        pagesOrError =
        pcall(function()

            return
                AvatarEditorService:
                SearchCatalogAsync(
                    params
                )

        end)

    if not searchOk
        or not pagesOrError then

        state.loading = false

        ResultStatus.Text =
            "Search failed"

        EmptyLabel.Text =
            "Roblox rejected the catalog search.\n\n"
            .. shorten(
                tostring(
                    pagesOrError
                ),
                220
            )

        return
    end

    state.pagesObject =
        pagesOrError

    --------------------------------------------------------
    -- Get first page
    --------------------------------------------------------

    local pageOk,
        itemsOrError =
        pcall(function()

            return
                state.pagesObject:
                GetCurrentPage()

        end)

    if not pageOk
        or type(itemsOrError)
            ~= "table" then

        state.loading = false

        ResultStatus.Text =
            "Page failed"

        EmptyLabel.Text =
            shorten(
                tostring(
                    itemsOrError
                ),
                220
            )

        return
    end

    local items =
        itemsOrError

    state.pageCache[1] =
        items

    state.loading = false

    displayPage(
        items,
        1
    )
end

------------------------------------------------------------
-- NEXT PAGE
------------------------------------------------------------

local function nextPage()

    if state.loading then
        return
    end

    if not state.pagesObject then
        return
    end

    --------------------------------------------------------
    -- Already cached
    --------------------------------------------------------

    local cached =
        state.pageCache[
            state.page + 1
        ]

    if cached then

        displayPage(
            cached,
            state.page + 1
        )

        return
    end

    --------------------------------------------------------
    -- Finished
    --------------------------------------------------------

    if state.pagesObject.IsFinished then
        return
    end

    state.loading = true

    ResultStatus.Text =
        "Loading page..."

    local ok,
        errorMessage =
        pcall(function()

            state.pagesObject:
                AdvanceToNextPageAsync()

        end)

    if not ok then

        state.loading = false

        ResultStatus.Text =
            "Page failed"

        EmptyLabel.Visible =
            true

        EmptyLabel.Text =
            shorten(
                tostring(
                    errorMessage
                ),
                220
            )

        return
    end

    local pageOk,
        items =
        pcall(function()

            return
                state.pagesObject:
                GetCurrentPage()

        end)

    if not pageOk
        or type(items)
            ~= "table" then

        state.loading = false

        ResultStatus.Text =
            "Page failed"

        EmptyLabel.Visible =
            true

        EmptyLabel.Text =
            tostring(
                items
            )

        return
    end

    local newPage =
        state.page + 1

    state.pageCache[newPage] =
        items

    state.loading = false

    displayPage(
        items,
        newPage
    )
end

local dragging = false
local dragStart
local windowStart

Header.InputBegan:Connect(function(input)

    if
        input.UserInputType
            == Enum.UserInputType.MouseButton1

        or input.UserInputType
            == Enum.UserInputType.Touch
    then

        dragging = true

        dragStart =
            input.Position

        windowStart =
            Window.Position

        input.Changed:Connect(function()

            if
                input.UserInputState
                    == Enum.UserInputState.End
            then

                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)

    if not dragging then
        return
    end

    if
        input.UserInputType
            ~= Enum.UserInputType.MouseMovement

        and input.UserInputType
            ~= Enum.UserInputType.Touch
    then

        return
    end

    local delta =
        input.Position
        - dragStart

    local newX =
        windowStart.X.Offset
        + delta.X

    local newY =
        windowStart.Y.Offset
        + delta.Y

    local cam =
        workspace.CurrentCamera

    if cam then

        local size =
            cam.ViewportSize

        local windowSize =
            Window.AbsoluteSize

        newX =
            math.clamp(
                newX,
                4,
                math.max(
                    4,
                    size.X
                    - windowSize.X
                    - 4
                )
            )

        newY =
            math.clamp(
                newY,
                4,
                math.max(
                    4,
                    size.Y
                    - windowSize.Y
                    - 4
                )
            )
    end

    Window.Position =
        UDim2.fromOffset(
            newX,
            newY
        )
end)

-- CLAMP WINDOW
------------------------------------------------------------

local function clampWindow()

    local cam =
        workspace.CurrentCamera

    if not cam then
        return
    end

    local size =
        cam.ViewportSize

    local targetWidth =
        math.min(
            Window.AbsoluteSize.X,
            math.max(
                120,
                size.X - 8
            )
        )

    local targetHeight

    if state.minimized then

        targetHeight =
            CONFIG.HeaderHeight

    else

        targetHeight =
            math.min(
                Window.AbsoluteSize.Y,
                math.max(
                    60,
                    size.Y - 8
                )
            )
    end

    Window.Size =
        UDim2.fromOffset(
            targetWidth,
            targetHeight
        )

    local x =
        Window.Position.X.Offset

    local y =
        Window.Position.Y.Offset

    x =
        math.clamp(
            x,
            4,
            math.max(
                4,
                size.X
                - targetWidth
                - 4
            )
        )

    y =
        math.clamp(
            y,
            4,
            math.max(
                4,
                size.Y
                - targetHeight
                - 4
            )
        )

    Window.Position =
        UDim2.fromOffset(
            x,
            y
        )
end

    local columns =
        math.floor(
            resultWidth
            / (
                CONFIG.CardWidth
                + 6
            )
        )

    columns =
        math.max(
            1,
            columns
        )

    local available =
        resultWidth
        - 10
        - (
            (columns - 1)
            * 6
        )

    local cardWidth =
        math.floor(
            available
            / columns
        )

    cardWidth =
        math.max(
            95,
            math.min(
                CONFIG.CardWidth,
                cardWidth
            )
        )

    GridLayout.CellSize =
        UDim2.fromOffset(
            cardWidth,
            CONFIG.CardHeight
        )
end

Window:GetPropertyChangedSignal(
    "AbsoluteSize"
):Connect(function()

    updateGrid()
end)

ResultsArea:GetPropertyChangedSignal(
    "AbsoluteSize"
):Connect(function()

    updateGrid()
end)

        clampWindow()
        updateGrid()

        task.wait(
            0.5

    end
