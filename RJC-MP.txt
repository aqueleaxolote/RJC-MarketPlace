local function create(className, properties)
    local object = Instance.new(className)

    for property, value in pairs(properties or {}) do
        pcall(function()
            object[property] = value
        end)
    end

    return object
end

local function corner(parent, radius)
    return create("UICorner", {
        CornerRadius = UDim.new(0, radius or 7),
        Parent = parent,
    })
end

local function outline(parent, color, thickness)
    return create("UIStroke", {
        Color = color or COLORS.Border,
        Thickness = thickness or 1,
        Transparency = 0.18,
        Parent = parent,
    })
end

local function shorten(text, length)
    text = tostring(text or "")

    if #text <= length then
        return text
    end

    return text:sub(1, length - 3) .. "..."
end

local function formatNumber(number)
    number = tonumber(number)

    if not number then
        return "?"
    end

    local result = tostring(math.floor(number))

    while true do
        local replacement, count =
            result:gsub(
                "^(-?%d+)(%d%d%d)",
                "%1,%2"
            )

        result = replacement

        if count == 0 then
            break
        end
    end

    return result
end

local function copy(text)

    local methods = {
        function()
            if setclipboard then
                setclipboard(text)
                return true
            end
        end,

        function()
            if toclipboard then
                toclipboard(text)
                return true
            end
        end,

        function()
            if set_clipboard then
                set_clipboard(text)
                return true
            end
        end,
    }

    for _, method in ipairs(methods) do
        local ok, result = pcall(method)

        if ok and result then
            return true
        end
    end

    return false
end

local function getEnumItem(enumObject, candidates)

    if not enumObject then
        return nil
    end

    for _, name in ipairs(candidates) do

        local ok, value =
            pcall(function()
                return enumObject[name]
            end)

        if ok and value then
            return value
        end
    end

    return nil
end

local function assetType(name)
    return getEnumItem(
        Enum.AvatarAssetType,
        {name}
    )
end

local function bundleType(name)
    return getEnumItem(
        Enum.BundleType,
        {name}
    )
end

local function categoryFilter(name)
    return getEnumItem(
        Enum.CatalogCategoryFilter,
        {name}
    )
end

local function sortEnum(...)
    return getEnumItem(
        Enum.CatalogSortType,
        {...}
    )
end

local function creatorEnum(name)
    return getEnumItem(
        Enum.CreatorTypeFilter,
        {name}
    )
end

local function assetTypes(...)
    local result = {}

    for _, name in ipairs({...}) do
        local value = assetType(name)

        if value then
            table.insert(result, value)
        end
    end

    return result
end

local function bundleTypes(...)
    local result = {}

    for _, name in ipairs({...}) do
        local value = bundleType(name)

        if value then
            table.insert(result, value)
        end
    end

    return result
end

-- DROPDOWN
------------------------------------------------------------

local function makeDropdown(
    title,
    options,
    getIndex,
    setIndex
)

    local holder =
        create(
            "Frame",
            {
                BackgroundTransparency = 1,

                Size =
                    UDim2.new(
                        1,
                        -2,
                        0,
                        49
                    ),

                Parent =
                    SidebarScroll,
            }
        )

    create(
        "TextLabel",
        {
            BackgroundTransparency = 1,

            Position =
                UDim2.fromOffset(
                    1,
                    0
                ),

            Size =
                UDim2.new(
                    1,
                    -2,
                    0,
                    15
                ),

            Font =
                Enum.Font.GothamBold,

            Text =
                title:upper(),

            TextColor3 =
                COLORS.SubText,

            TextSize = 9,

            TextXAlignment =
                Enum.TextXAlignment.Left,

            Parent =
                holder,
        }
    )

    local button =
        create(
            "TextButton",
            {
                BackgroundColor3 =
                    COLORS.Panel2,

                BorderSizePixel = 0,

                Position =
                    UDim2.fromOffset(
                        0,
                        17
                    ),

                Size =
                    UDim2.new(
                        1,
                        0,
                        0,
                        31
                    ),

                Font =
                    Enum.Font.Gotham,

                Text = "",

                TextColor3 =
                    COLORS.Text,

                TextSize = 10,

                TextXAlignment =
                    Enum.TextXAlignment.Left,

                AutoButtonColor = false,

                Parent =
                    holder,
            }
        )

    corner(button, 6)
    outline(button)

    create(
        "TextLabel",
        {
            BackgroundTransparency = 1,

            Position =
                UDim2.new(
                    1,
                    -23,
                    0,
                    0
                ),

            Size =
                UDim2.fromOffset(
                    20,
                    31
                ),

            Font =
                Enum.Font.GothamBold,

            Text =
                "v",

            TextColor3 =
                COLORS.SubText,

            TextSize = 10,

            Parent =
                button,
        }
    )

    local function optionName(option)

        return tostring(
            option.name
            or option[1]
            or ""
        )
    end

    local function refresh()

        local index =
            getIndex()

        local option =
            options[index]

        if option then
            button.Text =
                "  "
                .. optionName(option)
        end
    end

    button.Activated:Connect(
        function()

            closeDropdown()

            local menuHeight =
                math.min(
                    250,
                    10 + (#options * 31)
                )

            local menu =
                create(
                    "ScrollingFrame",
                    {
                        BackgroundColor3 =
                            COLORS.Panel2,

                        BorderSizePixel = 0,

                        Size =
                            UDim2.fromOffset(
                                math.max(
                                    140,
                                    button.AbsoluteSize.X
                                ),
                                menuHeight
                            ),

                        CanvasSize =
                            UDim2.new(
                                0,
                                0,
                                0,
                                0
                            ),

                        AutomaticCanvasSize =
                            Enum.AutomaticSize.Y,

                        ScrollBarThickness = 2,

                        ScrollBarImageColor3 =
                            COLORS.Accent,

                        ZIndex = 1000,

                        Parent =
                            DropdownLayer,
                    }
                )

            corner(menu, 7)
            outline(menu)

            activeDropdown =
                menu

            create(
                "UIPadding",
                {
                    PaddingTop =
                        UDim.new(
                            0,
                            5
                        ),

                    PaddingBottom =
                        UDim.new(
                            0,
                            5
                        ),

                    Parent =
                        menu,
                }
            )

            create(
                "UIListLayout",
                {
                    Padding =
                        UDim.new(
                            0,
                            2
                        ),

                    Parent =
                        menu,
                }
            )

            for index, option in ipairs(options) do

                local selected =
                    index == getIndex()

                local optionButton =
                    create(
                        "TextButton",
                        {
                            BackgroundColor3 =
                                selected
                                and COLORS.AccentDark
                                or COLORS.Panel2,

                            BackgroundTransparency =
                                selected
                                and 0
                                or 1,

                            BorderSizePixel = 0,

                            Size =
                                UDim2.new(
                                    1,
                                    -8,
                                    0,
                                    28
                                ),

                            Font =
                                Enum.Font.Gotham,

                            Text =
                                "  "
                                .. optionName(option),

                            TextColor3 =
                                COLORS.Text,

                            TextSize = 9,

                            TextXAlignment =
                                Enum.TextXAlignment.Left,

                            AutoButtonColor = false,

                            ZIndex = 1001,

                            Parent =
                                menu,
                        }
                    )

                corner(optionButton, 5)

                optionButton.Activated:Connect(
                    function()

                        setIndex(
                            index,
                            option
                        )

                        refresh()

                        closeDropdown()
                    end
                )

                optionButton.MouseEnter:Connect(
                    function()

                        if index ~= getIndex() then

                            optionButton.BackgroundColor3 =
                                COLORS.Hover

                            optionButton.BackgroundTransparency =
                                0
                        end
                    end
                )

                optionButton.MouseLeave:Connect(
                    function()

                        if index ~= getIndex() then
                            optionButton.BackgroundTransparency =
                                1
                        end
                    end
                )
            end

            task.defer(
                function()

                    if not menu.Parent then
                        return
                    end

                    local cam =
                        workspace.CurrentCamera

                    local screen =
                        cam
                        and cam.ViewportSize
                        or Vector2.new(
                            800,
                            600
                        )

                    local x =
                        button.AbsolutePosition.X

                    local y =
                        button.AbsolutePosition.Y
                        + button.AbsoluteSize.Y
                        + 2

                    local w =
                        menu.AbsoluteSize.X

                    local h =
                        menu.AbsoluteSize.Y

                    x =
                        math.clamp(
                            x,
                            3,
                            math.max(
                                3,
                                screen.X
                                - w
                                - 3
                            )
                        )

                    y =
                        math.clamp(
                            y,
                            3,
                            math.max(
                                3,
                                screen.Y
                                - h
                                - 3
                            )
                        )

                    menu.Position =
                        UDim2.fromOffset(
                            x,
                            y
                        )
                end
            )
        end
    )

    refresh()

    return holder
end

local ThumbnailQueue = {}
local ThumbnailWorkerRunning = false

local function getItemID(item)
    return tonumber(
        item.Id
        or item.id
    )
end

local function isBundle(item)

    local value =
        tostring(
            item.ItemType
            or item.itemType
            or ""
        ):lower()

    return
        value == "bundle"
        or value == "2"
end

local function thumbnailContent(item)

    local id =
        getItemID(item)

    if not id then
        return nil
    end

    if isBundle(item) then

        return
            "rbxthumb://type=BundleThumbnail&id="
            .. tostring(id)
            .. "&w=420&h=420"
    end

    return
        "rbxthumb://type=Asset&id="
        .. tostring(id)
        .. "&w=420&h=420"
end

local function loadThumbnail(
    image,
    item
)

    if not image
        or not image.Parent then
        return
    end

    local content =
        thumbnailContent(item)

    if not content then
        return
    end

    image.Image =
        content

    task.spawn(
        function()

            for attempt = 1,
                CONFIG.ThumbnailRetries do

                if not image.Parent then
                    return
                end

                pcall(
                    function()

                        ContentProvider:PreloadAsync(
                            {
                                image
                            }
                        )

                    end
                )

                task.wait(
                    CONFIG.ThumbnailRetryDelay
                )

                if image.Parent then
                    image.Image =
                        content
                end
            end
        end
    )
end

local function startThumbnailWorker()

    if ThumbnailWorkerRunning then
        return
    end

    ThumbnailWorkerRunning = true

    task.spawn(
        function()

            while #ThumbnailQueue > 0 do

                local request =
                    table.remove(
                        ThumbnailQueue,
                        1
                    )

                if request
                    and request.image
                    and request.image.Parent then

                    loadThumbnail(
                        request.image,
                        request.item
                    )
                end

                task.wait(
                    CONFIG.ThumbnailQueueDelay
                )
            end

            ThumbnailWorkerRunning = false
        end
    )
end

local function queueThumbnail(
    image,
    item
)

    table.insert(
        ThumbnailQueue,
        {
            image = image,
            item = item,
        }
    )

    startThumbnailWorker()
end

local function getPriceText(item)

    local status =
        tostring(
            item.PriceStatus
            or item.priceStatus
            or ""
        )

    if status == "Free" then
        return "FREE"
    end

    if status == "OffSale" then
        return "OFFSALE"
    end

    local lowest =
        tonumber(
            item.LowestPrice
            or item.lowestPrice
        )

    if lowest
        and lowest > 0 then

        return
            "from "
            .. formatNumber(lowest)
            .. " R$"
    end

    local price =
        tonumber(
            item.Price
            or item.price
        )

    if price then

        if price == 0 then
            return "FREE"
        end

        return
            formatNumber(price)
            .. " R$"
    end

    return "PRICE N/A"
end

-- BUILD PARAMETERS

local function buildSearchParameters()

    local params =
        CatalogSearchParams.new()

    params.SearchKeyword =
        tostring(
            SearchBox.Text
            or ""
        )

    params.Limit =
        CONFIG.ResultsPerPage

    params.IncludeOffSale =
        false

    --------------------------------------------------------
    -- CATEGORY
    --------------------------------------------------------

    local category =
        CATEGORIES[
            state.categoryIndex
        ]

    if category then

        local filter =
            categoryFilter(
                category.category
            )

        if filter then
            params.CategoryFilter =
                filter
        end
    end

    --------------------------------------------------------
    -- SUBCATEGORY
    --------------------------------------------------------

    if category then

        local subcategory =
            category.subcategories[
                state.subcategoryIndex
            ]

        if subcategory then

            if subcategory.assetTypes
                and #subcategory.assetTypes > 0 then

                params.AssetTypes =
                    subcategory.assetTypes
            end

            if subcategory.bundleTypes
                and #subcategory.bundleTypes > 0 then

                params.BundleTypes =
                    subcategory.bundleTypes
            end
        end
    end

    --------------------------------------------------------
    -- PRICE
    --------------------------------------------------------

    local price =
        PRICES[
            state.priceIndex
        ]

    if price then

        if price.value ==
            "custom" then

            local custom =
                tonumber(
                    CustomPrice.Text
                )

            if custom then

                params.MaxPrice =
                    math.max(
                        0,
                        math.floor(custom)
                    )
            end

        elseif price.value ~= nil then

            params.MaxPrice =
                price.value
        end
    end

    --------------------------------------------------------
    -- SORT
    --------------------------------------------------------

    local sort =
        SORTS[
            state.sortIndex
        ]

    if sort
        and sort.enum then

        params.SortType =
            sort.enum
    end

    --------------------------------------------------------
    -- CREATOR
    --------------------------------------------------------

    local creatorText =
        tostring(
            CreatorBox.Text
            or ""
        )

    creatorText =
        creatorText:gsub(
            "^%s+",
            ""
        ):gsub(
            "%s+$",
            ""
        )

    if creatorText ~= "" then

        local creatorOption =
            CREATOR_TYPES[
                state.creatorTypeIndex
            ]

        local creatorType =
            creatorOption
            and creatorOption.enum

        local numericID =
            tonumber(creatorText)

        if numericID then

            if
                not creatorType
                or state.creatorTypeIndex == 3
            then

                error(
                    "Use User or Group for a creator ID."
                )
            end

            params.CreatorId =
                math.floor(
                    numericID
                )

            params.CreatorType =
                creatorType

        else

            params.CreatorName =
                creatorText

            if creatorType then
                params.CreatorType =
                    creatorType
            end
        end
    end

    return params
end

local function searchMarketplace()

    if state.loading then
        return
    end

    state.loading =
        true

    closeDropdown()
    hideDetails()
    clearGrid()

    ResultStatus.Text =
        "Searching..."

    EmptyLabel.Visible =
        true

    EmptyLabel.Text =
        "Searching Marketplace..."

    state.pagesObject =
        nil

    state.pageCache =
        {}

    state.page =
        1

    local params

    local parameterOK,
        parameterResult =
        pcall(
            function()
                return buildSearchParameters()
            end
        )

    if not parameterOK then

        state.loading =
            false

        ResultStatus.Text =
            "Search setup failed"

        EmptyLabel.Text =
            shorten(
                tostring(
                    parameterResult
                ),
                240
            )

        return
    end

    params =
        parameterResult

    local searchOK,
        pagesOrError =
        pcall(
            function()

                return
                    AvatarEditorService:
                    SearchCatalogAsync(
                        params
                    )
            end
        )

    if not searchOK
        or not pagesOrError then

        state.loading =
            false

        ResultStatus.Text =
            "Search failed"

        EmptyLabel.Text =
            "Roblox rejected the catalog search.\n\n"
            .. shorten(
                tostring(
                    pagesOrError
                ),
                260
            )

        return
    end

    state.pagesObject =
        pagesOrError

    local pageOK,
        itemsOrError =
        pcall(
            function()

                return
                    state.pagesObject:
                    GetCurrentPage()
            end
        )

    if not pageOK
        or type(itemsOrError) ~= "table" then

        state.loading =
            false

        ResultStatus.Text =
            "Page failed"

        EmptyLabel.Text =
            shorten(
                tostring(
                    itemsOrError
                ),
                240
            )

        return
    end

    state.pageCache[1] =
        itemsOrError

    state.loading =
        false

    displayPage(
        itemsOrError,
        1
    )
end

local function nextPage()

    if state.loading then
        return
    end

    if not state.pagesObject then
        return
    end

    local nextNumber =
        state.page + 1

    local cached =
        state.pageCache[
            nextNumber
        ]

    if cached then

        displayPage(
            cached,
            nextNumber
        )

        return
    end

    if state.pagesObject.IsFinished then
        return
    end

    state.loading =
        true

    ResultStatus.Text =
        "Loading page..."

    local advanceOK,
        advanceError =
        pcall(
            function()

                state.pagesObject:
                    AdvanceToNextPageAsync()
            end
        )

    if not advanceOK then

        state.loading =
            false

        ResultStatus.Text =
            "Page failed"

        EmptyLabel.Visible =
            true

        EmptyLabel.Text =
            shorten(
                tostring(
                    advanceError
                ),
                240
            )

        return
    end

    local pageOK,
        items =
        pcall(
            function()

                return
                    state.pagesObject:
                    GetCurrentPage()
            end
        )

    if not pageOK
        or type(items) ~= "table" then

        state.loading =
            false

        ResultStatus.Text =
            "Page failed"

        EmptyLabel.Visible =
            true

        EmptyLabel.Text =
            shorten(
                tostring(items),
                240
            )

        return
    end

    state.pageCache[
        nextNumber
    ] = items

    state.loading =
        false

    displayPage(
        items,
        nextNumber
    )
end

-- DRAGGING

local dragging = false
local dragStart
local windowStart

Header.InputBegan:Connect(
    function(input)

        if
            input.UserInputType
                == Enum.UserInputType.MouseButton1

            or input.UserInputType
                == Enum.UserInputType.Touch
        then

            dragging = true

            dragStart =
                input.Position

            windowStart =
                Window.Position

            input.Changed:Connect(
                function()

                    if
                        input.UserInputState
                            == Enum.UserInputState.End
                    then

                        dragging = false
                    end
                end
            )
        end
    end
)

UserInputService.InputChanged:Connect(
    function(input)

        if not dragging then
            return
        end

        if
            input.UserInputType
                ~= Enum.UserInputType.MouseMovement
            and input.UserInputType
                ~= Enum.UserInputType.Touch
        then

            return
        end

        local delta =
            input.Position
            - dragStart

        local newX =
            windowStart.X.Offset
            + delta.X

        local newY =
            windowStart.Y.Offset
            + delta.Y

        local cam =
            workspace.CurrentCamera

        if cam then

            local screen =
                cam.ViewportSize

            local windowSize =
                Window.AbsoluteSize

            newX =
                math.clamp(
                    newX,
                    4,
                    math.max(
                        4,
                        screen.X
                        - windowSize.X
                        - 4
                    )
                )

            newY =
                math.clamp(
                    newY,
                    4,
                    math.max(
                        4,
                        screen.Y
                        - windowSize.Y
                        - 4
                    )
                )
        end

        Window.Position =
            UDim2.fromOffset(
                newX,
                newY
            )
    end
)

-- CLAMP WINDOW
------------------------------------------------------------

local function clampWindow()

    local cam =
        workspace.CurrentCamera

    if not cam then
        return
    end

    local screen =
        cam.ViewportSize

    local targetWidth =
        math.min(
            Window.AbsoluteSize.X,
            math.max(
                120,
                screen.X - 8
            )
        )

    local targetHeight

    if state.minimized then

        targetHeight =
            CONFIG.HeaderHeight

    else

        targetHeight =
            math.min(
                Window.AbsoluteSize.Y,
                math.max(
                    60,
                    screen.Y - 8
                )
            )
    end

    Window.Size =
        UDim2.fromOffset(
            targetWidth,
            targetHeight
        )

    local x =
        Window.Position.X.Offset

    local y =
        Window.Position.Y.Offset

    x =
        math.clamp(
            x,
            4,
            math.max(
                4,
                screen.X
                - targetWidth
                - 4
            )
        )

    y =
        math.clamp(
            y,
            4,
            math.max(
                4,
                screen.Y
                - targetHeight
                - 4
            )
        )

    Window.Position =
        UDim2.fromOffset(
            x,
            y
        )
end

------------------------------------------------------------
-- MINIMIZE
------------------------------------------------------------

local function setMinimized(value)

    state.minimized =
        value

    closeDropdown()

    if value then

        state.savedHeight =
            Window.AbsoluteSize.Y

        Body.Visible =
            false

        Window.Size =
            UDim2.fromOffset(
                Window.AbsoluteSize.X,
                CONFIG.HeaderHeight
            )

        MinimizeButton.Text =
            "+"

    else

        Body.Visible =
            true

        Window.Size =
            UDim2.fromOffset(
                Window.AbsoluteSize.X,
                math.max(
                    CONFIG.MinHeight,
                    math.min(
                        state.savedHeight,
                        CONFIG.Height
                    )
                )
            )

        MinimizeButton.Text =
            "-"
    end

    clampWindow()
end

MinimizeButton.Activated:Connect(
    function()

        setMinimized(
            not state.minimized
        )
    end
)

------------------------------------------------------------
-- OUTSIDE DROPDOWN CLICK
------------------------------------------------------------

UserInputService.InputBegan:Connect(
    function(input)

        if
            input.UserInputType
                ~= Enum.UserInputType.MouseButton1

            and input.UserInputType
                ~= Enum.UserInputType.Touch
        then

            return
        end

        if not activeDropdown then
            return
        end

        local point =
            input.Position

        local position =
            activeDropdown.AbsolutePosition

        local size =
            activeDropdown.AbsoluteSize

        local inside =
            point.X >= position.X
            and point.X
                <= position.X + size.X
            and point.Y >= position.Y
            and point.Y
                <= position.Y + size.Y

        if not inside then
            closeDropdown()
        end
    end
)

------------------------------------------------------------
-- RESPONSIVE GRID
------------------------------------------------------------

local function updateGrid()

    local width =
        ResultsArea.AbsoluteSize.X

    if width <= 0 then
        return
    end

    local columns =
        math.floor(
            width
            / (
                CONFIG.CardWidth
                + 7
            )
        )

    columns =
        math.max(
            1,
            columns
        )

    local available =
        width
        - 12
        - (
            (columns - 1)
            * 7
        )

    local cardWidth =
        math.floor(
            available
            / columns
        )

    cardWidth =
        math.max(
            100,
            math.min(
                CONFIG.CardWidth,
                cardWidth
            )
        )

    GridLayout.CellSize =
        UDim2.fromOffset(
            cardWidth,
            CONFIG.CardHeight
        )
end

Window:GetPropertyChangedSignal(
    "AbsoluteSize"
):Connect(
    function()
        updateGrid()
    end
)

ResultsArea:GetPropertyChangedSignal(
    "AbsoluteSize"
):Connect(
    function()
        updateGrid()
    end
)

------------------------------------------------------------
-- CONTINUOUS WINDOW MONITOR
------------------------------------------------------------

task.spawn(
    function()

        while ScreenGui.Parent do

            clampWindow()
            updateGrid()

            task.wait(0.5)
        end
    end
)

task.defer(
    function()

        updateGrid()
        clampWindow()

    end
